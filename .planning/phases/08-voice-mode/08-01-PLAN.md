```typescript
// src/app/api/tts/route.ts
import { ElevenLabsClient } from '@elevenlabs/elevenlabs-js';
import { hasEnvVar } from '@/lib/env';

const ALLOWED_VOICE_ID = 'dMWVPH9DSxWOMrrrUso3'; // per D-02, locked

export async function POST(req: Request) {
  if (!hasEnvVar('ELEVENLABS_API_KEY')) {
    return new Response(
      JSON.stringify({ error: 'TTS not configured' }),
      { status: 503, headers: { 'Content-Type': 'application/json' } }
    );
  }

  let text: string;
  let voiceId: string;
  try {
    const body = await req.json() as { text: unknown; voiceId: unknown };
    text = typeof body.text === 'string' ? body.text.trim().slice(0, 500) : '';
    voiceId = typeof body.voiceId === 'string' ? body.voiceId : ALLOWED_VOICE_ID;
    // Security: allowlist voice ID per D-02 threat mitigation
    if (voiceId !== ALLOWED_VOICE_ID) voiceId = ALLOWED_VOICE_ID;
    if (!text) {
      return new Response(
        JSON.stringify({ error: 'text is required' }),
        { status: 400, headers: { 'Content-Type': 'application/json' } }
      );
    }
  } catch {
    return new Response(
      JSON.stringify({ error: 'Invalid request body' }),
      { status: 400, headers: { 'Content-Type': 'application/json' } }
    );
  }

  try {
    const client = new ElevenLabsClient({ apiKey: process.env.ELEVENLABS_API_KEY });
    const audioStream = await client.textToSpeech.stream(voiceId, {
      text,
      modelId: 'eleven_turbo_v2_5',  // per D-04
      outputFormat: 'mp3_44100_128',
    });

    const stream = new ReadableStream({
      async start(controller) {
        for await (const chunk of audioStream) {
          controller.enqueue(chunk);
        }
        controller.close();
      },
    });

    return new Response(stream, {
      headers: { 'Content-Type': 'audio/mpeg' },
    });
  } catch {
    return new Response(
      JSON.stringify({ error: 'TTS failed' }),
      { status: 500, headers: { 'Content-Type': 'application/json' } }
    );
  }
}
```

## Trust Boundaries

| Boundary | Description |
|----------|-------------|
| client → /api/tts | Untrusted text and voiceId inputs cross from browser to server; ElevenLabs API key must never cross back |
| server → ElevenLabs | Server-side key injected; audio bytes stream back to client |
| browser → window.VoiceBus | All components read/write this singleton; must be initialized before any React tree |

## STRIDE Threat Register

| Threat ID | Category | Component | Disposition | Mitigation Plan |
|-----------|----------|-----------|-------------|-----------------|
| T-08-01 | Information Disclosure | ELEVENLABS_API_KEY in /api/tts | mitigate | Server-only `process.env.ELEVENLABS_API_KEY`; `hasEnvVar()` guard; never `NEXT_PUBLIC_*`; route returns 503 when absent |
| T-08-02 | Tampering | voiceId field in POST /api/tts body | mitigate | Allowlist: any voiceId that is not `dMWVPH9DSxWOMrrrUso3` is silently overwritten to the locked value (D-02) |
| T-08-03 | Tampering | text field in POST /api/tts body — injection or oversized payload | mitigate | Validate: typeof === 'string', trim, slice to 500 chars max, reject empty string with 400 |
| T-08-04 | Denial of Service | Unbounded calls to /api/tts consuming ElevenLabs quota | accept | No rate limit in v1 (same pattern as /api/chat); public portfolio, low traffic; note for Phase 9 polish |
| T-08-05 | Elevation of Privilege | window.VoiceBus.attachMic() called without user gesture | mitigate | `getUserMedia` requires explicit user gesture to activate voice mode (AskParz button click); no auto-activation |
| T-08-06 | Tampering | VoiceBus global overwritten by third-party script | accept | No third-party scripts with write access to window; particles.js is the only window global and does not conflict |
